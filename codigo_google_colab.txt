import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
import warnings
from google.colab import files
import io

warnings.filterwarnings('ignore')

# Configuração visual premium para os gráficos ficarem de alto nível
plt.style.use('dark_background')
sns.set_palette("husl")

# 2. Upload Interativo (Específico para Google Colab)
print("Selecione a planilha de receitas do seu computador:")
uploaded = files.upload()

# Pegando o nome exato do arquivo que foi enviado
file_name = list(uploaded.keys())[0]

# 3. Carregando os Dados (Leitura Dinâmica Multiformato)
xls = pd.ExcelFile(io.BytesIO(uploaded[file_name]))

# Função auxiliar para rastrear as linhas na planilha estritamente dentro de um bloco
def extract_row_bounded(df, keyword, start_idx, end_idx):
    for r in range(start_idx, end_idx):
        row = df.iloc[r]
        for c, val in enumerate(row):
            if isinstance(val, str) and val.lower().strip() == keyword:
                data = []
                for i in range(c + 1, len(row)):
                    v = row.iloc[i]
                    if pd.notna(v):
                        if isinstance(v, str):
                            try: v = float(v.replace('R$', '').replace('.', '').replace(',', '.').strip())
                            except: continue
                        data.append(v)
                    if len(data) == 12: break
                while len(data) < 12: data.append(0)
                return data
    return [0]*12

# 4. Extraindo as métricas
total_antecipacao = 0
total_ro = 0
vendas_ano_mais_recente = []
ro_ano_mais_recente = []
antecip_ano_mais_recente = []
ultimo_ano = ""

if '7 - Receitas ' in xls.sheet_names:
    df = pd.read_excel(io.BytesIO(uploaded[file_name]), sheet_name='7 - Receitas ', header=None)
    year_anchors = []
    for r in range(len(df)):
        val = df.iloc[r, 1] # Coluna B
        if val in [2023, 2024, 2025, 2026, '2023', '2024', '2025', '2026']:
            year_anchors.append({'year': str(val), 'row': r})
            
    for i, anchor in enumerate(year_anchors):
        start_row = anchor['row']
        end_row = year_anchors[i+1]['row'] if i < len(year_anchors)-1 else len(df)
        year = anchor['year']
        ultimo_ano = year
        
        ro = extract_row_bounded(df, 'ro', start_row, end_row)
        if sum(ro) == 0: ro = extract_row_bounded(df, 'rlo', start_row, end_row)
        antecip = extract_row_bounded(df, 'taxa antecipação', start_row, end_row)
        
        total_antecipacao += sum(antecip)
        total_ro += sum(ro)
        
        if i == len(year_anchors) - 1:
            vendas_ano_mais_recente = extract_row_bounded(df, 'vendas', start_row, end_row)
            ro_ano_mais_recente = ro
            antecip_ano_mais_recente = antecip
else:
    sheet_names = [s for s in xls.sheet_names if str(s).startswith('Receita ')]
    if sheet_names:
        ultimo_ano = sheet_names[-1].replace('Receita ', '')
    for sheet in sheet_names:
        df = pd.read_excel(io.BytesIO(uploaded[file_name]), sheet_name=sheet, header=None)
        ro = extract_row_bounded(df, 'ro', 0, len(df))
        if sum(ro) == 0: ro = extract_row_bounded(df, 'rlo', 0, len(df))
        antecip = extract_row_bounded(df, 'taxa antecipação', 0, len(df))
        
        total_antecipacao += sum(antecip)
        total_ro += sum(ro)
        
        if sheet == sheet_names[-1]:
            vendas_ano_mais_recente = extract_row_bounded(df, 'vendas', 0, len(df))
            ro_ano_mais_recente = ro
            antecip_ano_mais_recente = antecip

meses = ['Jan', 'Fev', 'Mar', 'Abr', 'Mai', 'Jun', 'Jul', 'Ago', 'Set', 'Out', 'Nov', 'Dez']

# 5. O GRANDE ACHADO: Cálculo do Ralo Invisível
impacto_pct = (total_antecipacao / total_ro) * 100 if total_ro > 0 else 0

print("\n" + "="*50)
print(f"🚨 ALERTA DE DRENAGEM DE CAIXA 🚨")
print("="*50)
print(f"Total perdido em Taxas de Antecipação (Acumulado de todos os anos): R$ {total_antecipacao:,.2f}")
print(f"Isso consumiu {impacto_pct:.2f}% de todo o Lucro Operacional (RO) do período.")
print("="*50 + "\n")

# 6. Demonstração Visual 1: Gráfico de Drenagem (Ano Mais Recente)
plt.figure(figsize=(12, 5))
sns.barplot(x=meses, y=antecip_ano_mais_recente, color='red', alpha=0.8)
plt.title(f'Drenagem Mensal de Caixa: Taxas de Antecipação ({ultimo_ano})', fontsize=14, color='white')
plt.ylabel('Valor Perdido (R$)', color='white')
plt.xlabel('Meses', color='white')
plt.grid(axis='y', alpha=0.2)
plt.show()

print("\n")

# 7. Demonstração Visual 2: Vendas Brutas vs Lucro Real (Ano Mais Recente)
plt.figure(figsize=(12, 5))
plt.plot(meses, vendas_ano_mais_recente, marker='o', label='Faturamento Bruto (Vendas)', color='#3b82f6', linewidth=2)
plt.plot(meses, ro_ano_mais_recente, marker='s', label='Lucro Real (RO)', color='#10b981', linewidth=2)
plt.title(f'A Ilusão do Faturamento: Vendas Brutas vs Lucro que sobra no Caixa ({ultimo_ano})', fontsize=14, color='white')
plt.ylabel('Valores em R$', color='white')
plt.legend()
plt.grid(alpha=0.2)
plt.show()
